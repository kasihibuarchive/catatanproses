   307→/** Quiet link that renders the month recap as a washi PNG poster and downloads it. */
   308→function DownloadMonthImage({ month }: { month: MonthGroup }) {
   309→  const [busy, setBusy] = useState(false);
   310→
   311→  const download = async () => {
   312→    setBusy(true);
   313→    try {
   314→      const blob = await renderMonthSummaryBlob(month);
   315→      const url = URL.createObjectURL(blob);
   316→      const a = document.createElement("a");
   317→      a.href = url;
   318→      a.download = `kekiro-${month.key}.png`;
   319→      document.body.appendChild(a);
   320→      a.click();
   321→      a.remove();
   322→      window.setTimeout(() => URL.revokeObjectURL(url), 4000);
   323→      toast.success("Gambar ringkasan diunduh.");
   324→    } catch {
   325→      toast.error("Gagal membuat gambar.");
   326→    } finally {
   327→      setBusy(false);
   328→    }
   329→  };
   330→
   331→  return (
   332→    <button type="button" onClick={download} disabled={busy} className={textLink}>
   333→      {busy ? "menyiapkan…" : "gambar"}
   334→    </button>
   335→  );
   336→}

   369→function LogForm({
   370→  onCreated,
   371→  lastLog,
   372→}: {
   373→  onCreated: (log: PracticeLog) => void;
   374→  lastLog: PracticeLog | null;
   375→}) {
   376→  const [actorName, setActorName] = useState("");
   377→  const [title, setTitle] = useState("");
   378→  const [durationHours, setDurationHours] = useState("");
   379→  const [notes, setNotes] = useState("");
   380→  const [dateValue, setDateValue] = useState(""); // "" = today
   381→  const [showDatePicker, setShowDatePicker] = useState(false);
   382→  const [todayKey, setTodayKey] = useState("");
   383→  const [photo, setPhoto] = useState<File | null>(null);
   384→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   385→  const [submitting, setSubmitting] = useState(false);
   386→  const [fieldError, setFieldError] = useState<string | null>(null);
   387→  const fileRef = useRef<HTMLInputElement>(null);
   388→
   389→  // Prefill name from localStorage (client only).
   390→  useEffect(() => {
   391→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   392→    if (saved) setActorName(saved);
   393→    setTodayKey(toDateKey(new Date()));
   394→  }, []);
   395→
   396→  const isToday = !dateValue || (todayKey !== "" && dateValue === todayKey);
   397→  const isYesterday =
   398→    !isToday &&
   399→    todayKey !== "" &&
   400→    dateValue === toDateKey(new Date(Date.now() - 86400000));
   401→
   402→  // "Use last practice" is only useful while the form is still blank.
   403→  const canReuse = Boolean(lastLog) && !title && !durationHours && !notes;
   404→  const reuseLast = () => {
   405→    if (!lastLog) return;
   406→    setTitle(lastLog.title);
   407→    setDurationHours(minutesToHoursInput(lastLog.durationMin));
   408→    setNotes(lastLog.notes);
   409→  };
   410→
   411→  const pickedDateLabel =
   412→    dateValue && !isToday
   413→      ? new Date(`${dateValue}T00:00:00`).toLocaleDateString("id-ID", {
   414→          day: "numeric",
   415→          month: "long",
   416→        })
   417→      : "hari ini";
   418→
   419→  const pickPhoto = (file: File | null) => {
   420→    if (photoPreview) URL.revokeObjectURL(photoPreview);
   421→    setPhoto(file);
   422→    setPhotoPreview(file ? URL.createObjectURL(file) : null);
   423→  };
   424→
   425→  const reset = (keepName: boolean) => {
   426→    if (!keepName) setActorName("");
   427→    setTitle("");
   428→    setDurationHours("");
   429→    setNotes("");
   430→    setDateValue("");
   431→    setShowDatePicker(false);
   432→    pickPhoto(null);
   433→    if (fileRef.current) fileRef.current.value = "";
   434→  };
   435→
   436→  const submit = async (e?: React.FormEvent) => {
   437→    e?.preventDefault();
   438→    setFieldError(null);
   439→
   440→    if (!actorName.trim()) return setFieldError("Isi nama/divisi dulu ya.");
   441→    if (!title.trim()) return setFieldError("Tulis apa yang dilatih hari ini.");
   442→    const minutes = hoursToMinutes(durationHours);
   443→    if (minutes === null || minutes < 1) {
   444→      return setFieldError("Isi durasi latihan (jam).");
   445→    }
   446→    if (minutes > 1440) {
   447→      return setFieldError("Durasi maksimal 24 jam.");
   448→    }
   449→
   450→    setSubmitting(true);
   451→    try {
   452→      const fd = new FormData();
   453→      fd.set("actorName", actorName.trim());
   454→      fd.set("title", title.trim());
   455→      fd.set("durationMin", String(minutes));
   456→      fd.set("notes", notes.trim());
   457→      if (!isToday && dateValue) fd.set("date", dateValue);
   458→
   459→      let file: File | null = photo;
   460→      if (file) {
   461→        // Downscale in the browser so uploads stay small.
   462→        file = await downscaleImage(file);
   463→        if (file) fd.set("image", file);
   464→      }
   465→
   466→      const res = await fetch("/api/logs", { method: "POST", body: fd });
   467→      const data = (await res.json().catch(() => null)) as
   468→        | { log?: PracticeLog; error?: string }
   469→        | null;
   470→
   471→      if (!res.ok || !data?.log) {
   472→        throw new Error(data?.error ?? "Gagal menyimpan. Coba lagi.");
   473→      }
   474→
   475→      window.localStorage.setItem(ACTOR_NAME_STORAGE_KEY, actorName.trim());
   476→      onCreated(data.log);
   477→      reset(true);
   478→      toast.success("Latihan tercatat.");
   479→    } catch (err) {
   480→      setFieldError(
   481→        !navigator.onLine
   482→          ? "Kamu sedang offline. Sambungkan internet, lalu coba lagi."
   483→          : err instanceof Error
   484→            ? err.message
   485→            : "Gagal menyimpan. Coba lagi."
   486→      );
   487→    } finally {
   488→      setSubmitting(false);
   489→    }
   490→  };
   491→
   492→  return (
   493→    <form
   494→      onSubmit={submit}
   495→      onKeyDown={(e) => {
   496→        if (e.key === "Enter" && (e.metaKey || e.ctrlKey)) {
   497→          e.preventDefault();
   498→          void submit();
   499→        }
   500→      }}
   501→      className="pb-8 pt-10"
   502→      noValidate
   503→    >
   504→      <p
   505→        aria-hidden
   506→        className="font-kanji text-xs tracking-[0.35em] text-seal"
   507→      >
   508→        {isToday ? "今日の稽古" : isYesterday ? "昨日の稽古" : "過去の稽古"}
   509→      </p>
   510→      <h1 className="mt-2 font-serif text-2xl tracking-tight">
   511→        {isToday ? "Catat latihan hari ini" : `Catat latihan ${pickedDateLabel}`}
   512→      </h1>
   513→      <p className="mt-2 text-sm text-muted-foreground">
   514→        Isi detilnya, tambahkan foto kalau ada, lalu kirim.
   515→      </p>
   516→      {canReuse && (
   517→        <p className="mt-2 text-xs text-muted-foreground">
   518→          Latihan terakhir: “{lastLog?.title}” ·{" "}
   519→          <button type="button" onClick={reuseLast} className={textLink}>
   520→            pakai lagi
   521→          </button>
   522→        </p>
   523→      )}
   524→
   525→      {/* Date row — quiet by default, backdate on demand */}
   526→      <div className="mt-4 min-h-9">
   527→        {showDatePicker ? (
   528→          <div className="flex flex-wrap items-center gap-3">
   529→            <input
   530→              type="date"
   531→              value={dateValue}
   532→              max={todayKey || undefined}
   533→              onChange={(e) => setDateValue(e.target.value)}
   534→              aria-label="Tanggal latihan"
   535→              className="h-9 rounded-[3px] border border-foreground/15 bg-card/70 px-2 text-sm tabular-nums shadow-none outline-none transition-colors focus:border-seal"
   536→            />
   537→            <button
   538→              type="button"
   539→              onClick={() => setShowDatePicker(false)}
   540→              className={textLink}
   541→            >
   542→              selesai
   543→            </button>
   544→          </div>

   708→function relativeDayLabel(day: string): string | null {
   709→  const today = toDateKey(new Date());
   710→  if (day === today) return "Hari ini";
   711→  const target = new Date(`${day}T00:00:00`).getTime();
   712→  const now = new Date(`${today}T00:00:00`).getTime();
   713→  const diffDays = Math.round((now - target) / 86400000);
   714→  if (diffDays === 1) return "Kemarin";
   715→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   716→  return null;
   717→}

   719→function LogRow({
   720→  log,
   721→  justAdded,
   722→  onDeleted,
   723→  onUpdated,
   724→}: {
   725→  log: PracticeLog;
   726→  justAdded: boolean;
   727→  onDeleted: (id: string) => void;
   728→  onUpdated: (log: PracticeLog) => void;
   729→}) {
   730→  const [confirming, setConfirming] = useState(false);
   731→  const [editing, setEditing] = useState(false);
   732→  const [saving, setSaving] = useState(false);
   733→  const [editError, setEditError] = useState<string | null>(null);
   734→  const [draft, setDraft] = useState({
   735→    actorName: log.actorName,
   736→    title: log.title,
   737→    durationHours: minutesToHoursInput(log.durationMin),
   738→    notes: log.notes,
   739→    date: dateKey(log.date),
   740→  });
   741→
   742→  const startEdit = () => {
   743→    setDraft({
   744→      actorName: log.actorName,
   745→      title: log.title,
   746→      durationHours: minutesToHoursInput(log.durationMin),
   747→      notes: log.notes,
   748→      date: dateKey(log.date),
   749→    });
   750→    setEditError(null);
   751→    setConfirming(false);
   752→    setEditing(true);
   753→  };
   754→
   755→  const saveEdit = async () => {
   756→    const name = draft.actorName.trim();
   757→    const title = draft.title.trim();
   758→    const minutes = hoursToMinutes(draft.durationHours);
   759→    if (!name) return setEditError("Nama tidak boleh kosong.");
   760→    if (!title) return setEditError("Judul tidak boleh kosong.");
   761→    if (!draft.date) return setEditError("Tanggal wajib ada.");
   762→    if (minutes === null || minutes < 1) {
   763→      return setEditError("Isi durasi (jam).");
   764→    }
   765→    if (minutes > 1440) {
   766→      return setEditError("Durasi maksimal 24 jam.");
   767→    }
   768→
   769→    setSaving(true);
   770→    setEditError(null);
   771→    try {
   772→      const dateChanged = draft.date !== dateKey(log.date);
   773→      const res = await fetch(`/api/logs/${log.id}`, {
   774→        method: "PATCH",
   775→        headers: { "Content-Type": "application/json" },
   776→        body: JSON.stringify({
   777→          actorName: name,
   778→          title,
   779→          durationMin: minutes,
   780→          notes: draft.notes.trim(),
   781→          ...(dateChanged ? { date: draft.date } : {}),
   782→        }),
   783→      });
   784→      const data = (await res.json().catch(() => null)) as
   785→        | { log?: PracticeLog; error?: string }
   786→        | null;
   787→      if (!res.ok || !data?.log) {
   788→        throw new Error(data?.error ?? "Gagal menyimpan perubahan.");
   789→      }
   790→      onUpdated(data.log);
   791→      setEditing(false);
   792→      toast.success("Perubahan tersimpan.");
   793→    } catch (err) {
   794→      setEditError(err instanceof Error ? err.message : "Gagal menyimpan perubahan.");
   795→    } finally {
   796→      setSaving(false);
   797→    }
   798→  };
   799→
   800→  const remove = async () => {
   801→    if (!confirming) {
   802→      setConfirming(true);
   803→      return;
   804→    }
   805→    try {
   806→      const res = await fetch(`/api/logs/${log.id}`, { method: "DELETE" });
   807→      if (!res.ok) throw new Error();
   808→      onDeleted(log.id);
   809→      toast.success("Catatan dihapus.");
   810→    } catch {
   811→      toast.error("Gagal menghapus. Coba lagi.");
   812→      setConfirming(false);
   813→    }
   814→  };
   815→
   816→  const day = dateKey(log.date);
   817→  const dayLabel =
   818→    relativeDayLabel(day) ??
   819→    new Date(`${day}T00:00:00`).toLocaleDateString("id-ID", {
   820→      day: "numeric",
   821→      month: "short",
   822→    });
   823→  const weekday = WEEKDAY_KANJI[new Date(`${day}T00:00:00`).getDay()];
   824→
   825→  if (editing) {
   826→    return (
   827→      <li className="rounded-[3px] border border-foreground/30 bg-card p-4">
   828→        <div className="space-y-4">
   829→          <div className="grid gap-4 sm:grid-cols-2">
   830→            <input
   831→              value={draft.actorName}
   832→              onChange={(e) => setDraft({ ...draft, actorName: e.target.value })}
   833→              aria-label="Nama/Divisi"
   834→              className={underlineInput}
   835→              autoFocus
   836→            />
   837→            <input
   838→              value={draft.durationHours}
   839→              onChange={(e) => setDraft({ ...draft, durationHours: e.target.value })}
   840→              type="text"
   841→              inputMode="decimal"
   842→              autoComplete="off"
   843→              aria-label="Durasi (jam)"
   844→              className={`${underlineInput} tabular-nums`}
   845→            />
   846→          </div>
   847→          <input
   848→            value={draft.title}
   849→            onChange={(e) => setDraft({ ...draft, title: e.target.value })}
   850→            aria-label="Apa yang dilatih"
   851→            className={underlineInput}
   852→          />
   853→          {/* re-date support — mirrors the main form's quiet date row */}
   854→          <div className="flex flex-wrap items-center gap-3">
   855→            <label
   856→              htmlFor={`edit-date-${log.id}`}
   857→              className="text-xs text-muted-foreground"
   858→            >
   859→              Tanggal
   860→            </label>
   861→            <input
   862→              id={`edit-date-${log.id}`}
   863→              type="date"
   864→              value={draft.date}
   865→              max={toDateKey(new Date())}
   866→              onChange={(e) => setDraft({ ...draft, date: e.target.value })}
   867→              className="h-9 rounded-[3px] border border-foreground/15 bg-card/70 px-2 text-sm tabular-nums shadow-none outline-none transition-colors focus:border-seal"
   868→            />
   869→          </div>
   870→          <textarea
   871→            value={draft.notes}
   872→            onChange={(e) => setDraft({ ...draft, notes: e.target.value })}
   873→            rows={3}
   874→            aria-label="Catatan"
   875→            className="w-full resize-y rounded-[3px] border border-foreground/15 bg-transparent px-3 py-2 text-sm outline-none transition-colors focus:border-seal"
   876→          />
   877→          {editError && (
   878→            <p role="alert" className="text-sm text-destructive">
   879→              {editError}
   880→            </p>
   881→          )}
   882→          <div className="flex items-center justify-end gap-4">
   883→            <button type="button" onClick={() => setEditing(false)} className={textLink}>
   884→              batal
   885→            </button>
   886→            <button
   887→              type="button"
   888→              onClick={saveEdit}
   889→              disabled={saving}
   890→              className="rounded-sm text-xs font-semibold underline underline-offset-2 disabled:opacity-50 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring"
   891→            >
   892→              {saving ? "menyimpan…" : "simpan"}
   893→            </button>
   894→          </div>
   895→        </div>
   896→      </li>
   897→    );
   898→  }
   899→
   900→  return (
   901→    <li
   902→      id={`log-${log.id}`}
   903→      className={`rounded-[3px] border bg-card p-4 transition-colors hover:border-foreground/30 ${
   904→        justAdded ? "entry-flash border-seal/40" : "border-foreground/10"
   905→      }`}
   906→    >
   907→      <div className="flex items-baseline justify-between gap-3">
   908→        <p className="font-serif text-sm font-bold">{log.actorName}</p>
   909→        <p className="shrink-0 text-xs tabular-nums text-muted-foreground">
   910→          <span aria-hidden className="font-kanji mr-1.5 text-[11px] text-seal/75">
   911→            {weekday}
   912→          </span>
   913→          {dayLabel} · {formatDuration(log.durationMin)}
   914→        </p>
   915→      </div>
   916→      <p className="mt-1 text-sm">{log.title}</p>
   917→      {log.notes && (
   918→        <p className="mt-1.5 whitespace-pre-line text-sm text-muted-foreground">{log.notes}</p>
   919→      )}
   920→      {log.imagePath && (
   921→        <a
   922→          href={log.imagePath}
   923→          target="_blank"
   924→          rel="noreferrer"
   925→          className={`group mt-4 block ${
   926→            // Prints taped into a Showa photo album: a hairline white frame,

   928→            log.id.charCodeAt(0) % 2 === 0 ? "-rotate-[0.4deg]" : "rotate-[0.4deg]"

   959→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   960→async function downscaleImage(file: File): Promise<File | null> {
   961→  try {
   962→    const bitmap = await createImageBitmap(file);
   963→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   964→    const w = Math.round(bitmap.width * scale);
   965→    const h = Math.round(bitmap.height * scale);
   966→
   967→    const canvas = document.createElement("canvas");
   968→    canvas.width = w;
   969→    canvas.height = h;
   970→    const ctx = canvas.getContext("2d");
   971→    if (!ctx) return file;
   972→    ctx.drawImage(bitmap, 0, 0, w, h);
   973→    bitmap.close();
   974→
   975→    const blob = await new Promise<Blob | null>((resolve) =>
   976→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   977→    );
   978→    if (!blob) return file;
   979→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   980→      type: "image/jpeg",
   981→    });
   982→  } catch {
   983→    return file; // fall back to the original file
   984→  }
   985→}
