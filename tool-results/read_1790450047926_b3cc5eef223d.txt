   262→   262→function LogForm({ onCreated }: { onCreated: (log: PracticeLog) => void }) {

   265→   265→  const [durationHours, setDurationHours] = useState("");
   266→   266→  const [notes, setNotes] = useState("");
   267→   267→  const [dateValue, setDateValue] = useState(""); // "" = today
   268→   268→  const [showDatePicker, setShowDatePicker] = useState(false);
   269→   269→  const [todayKey, setTodayKey] = useState("");
   270→   270→  const [photo, setPhoto] = useState<File | null>(null);
   271→   271→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   272→   272→  const [submitting, setSubmitting] = useState(false);
   273→   273→  const [fieldError, setFieldError] = useState<string | null>(null);
   274→   274→  const fileRef = useRef<HTMLInputElement>(null);
   275→   275→
   276→   276→  // Prefill name from localStorage (client only).
   277→   277→  useEffect(() => {
   278→   278→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   279→   279→    if (saved) setActorName(saved);
   280→   280→    setTodayKey(toDateKey(new Date()));
   281→   281→  }, []);
   282→   282→
   283→   283→  const isToday = !dateValue || (todayKey !== "" && dateValue === todayKey);
   284→   284→  const isYesterday =
   285→   285→    !isToday &&
   286→   286→    todayKey !== "" &&
   287→   287→    dateValue === toDateKey(new Date(Date.now() - 86400000));

   290→   290→    dateValue && !isToday

   389→   389→      {/* Date row — quiet by default, backdate on demand */}

   403→   403→              onClick={() => setShowDatePicker(false)}

   547→   547→function relativeDayLabel(day: string): string | null {
   548→   548→  const today = toDateKey(new Date());
   549→   549→  if (day === today) return "Hari ini";
   550→   550→  const target = new Date(`${day}T00:00:00`).getTime();
   551→   551→  const now = new Date(`${today}T00:00:00`).getTime();
   552→   552→  const diffDays = Math.round((now - target) / 86400000);
   553→   553→  if (diffDays === 1) return "Kemarin";
   554→   554→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   555→   555→  return null;
   556→   556→}

   558→   558→function LogRow({
   559→   559→  log,
   560→   560→  justAdded,
   561→   561→  onDeleted,
   562→   562→  onUpdated,
   563→   563→}: {
   564→   564→  log: PracticeLog;
   565→   565→  justAdded: boolean;
   566→   566→  onDeleted: (id: string) => void;
   567→   567→  onUpdated: (log: PracticeLog) => void;
   568→   568→}) {
   569→   569→  const [confirming, setConfirming] = useState(false);
   570→   570→  const [editing, setEditing] = useState(false);
   571→   571→  const [saving, setSaving] = useState(false);
   572→   572→  const [editError, setEditError] = useState<string | null>(null);
   573→   573→  const [draft, setDraft] = useState({
   574→   574→    actorName: log.actorName,
   575→   575→    title: log.title,
   576→   576→    durationHours: minutesToHoursInput(log.durationMin),

   593→   593→    const name = draft.actorName.trim();

   762→   762→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   763→   763→async function downscaleImage(file: File): Promise<File | null> {
   764→   764→  try {
   765→   765→    const bitmap = await createImageBitmap(file);
   766→   766→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   767→   767→    const w = Math.round(bitmap.width * scale);
   768→   768→    const h = Math.round(bitmap.height * scale);
   769→   769→
   770→   770→    const canvas = document.createElement("canvas");
   771→   771→    canvas.width = w;
   772→   772→    canvas.height = h;
   773→   773→    const ctx = canvas.getContext("2d");
   774→   774→    if (!ctx) return file;
   775→   775→    ctx.drawImage(bitmap, 0, 0, w, h);
   776→   776→    bitmap.close();
   777→   777→
   778→   778→    const blob = await new Promise<Blob | null>((resolve) =>
   779→   779→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   780→   780→    );
   781→   781→    if (!blob) return file;
   782→   782→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   783→   783→      type: "image/jpeg",
   784→   784→    });
   785→   785→  } catch {
   786→   786→    return file; // fall back to the original file
   787→   787→  }
   788→   788→}
