   312→/** Quiet link that renders the month recap as a washi PNG poster and downloads it. */
   313→function DownloadMonthImage({ month }: { month: MonthGroup }) {
   314→  const [busy, setBusy] = useState(false);
   315→
   316→  const download = async () => {
   317→    setBusy(true);
   318→    try {
   319→      const blob = await renderMonthSummaryBlob(month);
   320→      const url = URL.createObjectURL(blob);
   321→      const a = document.createElement("a");
   322→      a.href = url;
   323→      a.download = `kekiro-${month.key}.png`;
   324→      document.body.appendChild(a);
   325→      a.click();
   326→      a.remove();
   327→      window.setTimeout(() => URL.revokeObjectURL(url), 4000);
   328→      toast.success("Gambar ringkasan diunduh.");
   329→    } catch {
   330→      toast.error("Gagal membuat gambar.");
   331→    } finally {
   332→      setBusy(false);
   333→    }
   334→  };
   335→
   336→  return (
   337→    <button type="button" onClick={download} disabled={busy} className={textLink}>
   338→      {busy ? "menyiapkan…" : "gambar"}
   339→    </button>
   340→  );
   341→}

   343→/** Quiet link that downloads the month ledger as a plain CSV file. */
   344→function DownloadMonthCsv({ month }: { month: MonthGroup }) {
   345→  const download = () => {
   346→    const csv = `\uFEFF${buildMonthCsv(month)}`;
   347→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   348→    const url = URL.createObjectURL(blob);
   349→    const a = document.createElement("a");
   350→    a.href = url;
   351→    a.download = `kekiro-${month.key}.csv`;
   352→    document.body.appendChild(a);
   353→    a.click();
   354→    a.remove();
   355→    window.setTimeout(() => URL.revokeObjectURL(url), 4000);
   356→    toast.success("CSV diunduh.");
   357→  };
   358→
   359→  return (
   360→    <button type="button" onClick={download} className={textLink}>
   361→      csv
   362→    </button>
   363→  );

   397→function LogForm({
   398→  onCreated,
   399→  lastLog,
   400→}: {
   401→  onCreated: (log: PracticeLog) => void;
   402→  lastLog: PracticeLog | null;
   403→}) {
   404→  const [actorName, setActorName] = useState("");
   405→  const [title, setTitle] = useState("");
   406→  const [durationHours, setDurationHours] = useState("");
   407→  const [notes, setNotes] = useState("");
   408→  const [dateValue, setDateValue] = useState(""); // "" = today
   409→  const [showDatePicker, setShowDatePicker] = useState(false);
   410→  const [todayKey, setTodayKey] = useState("");
   411→  const [photo, setPhoto] = useState<File | null>(null);
   412→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   413→  const [submitting, setSubmitting] = useState(false);
   414→  const [fieldError, setFieldError] = useState<string | null>(null);
   415→  const [draftRestored, setDraftRestored] = useState(false);
   416→  const fileRef = useRef<HTMLInputElement>(null);
   417→
   418→  // Prefill name from localStorage (client only), then restore any unsent
   419→  // draft so a reload never loses what was being typed.
   420→  useEffect(() => {
   421→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   422→    if (saved) setActorName(saved);
   423→    setTodayKey(toDateKey(new Date()));
   424→
   425→    try {
   426→      const raw = window.localStorage.getItem(DRAFT_STORAGE_KEY);
   427→      if (!raw) return;
   428→      const d = JSON.parse(raw) as Partial<FormDraft>;
   429→      if (!d.title && !d.durationHours && !d.notes) return;
   430→      if (d.actorName) setActorName(d.actorName);
   431→      if (d.title) setTitle(d.title);
   432→      if (d.durationHours) setDurationHours(d.durationHours);
   433→      if (d.notes) setNotes(d.notes);
   434→      setDraftRestored(true);
   435→    } catch {
   436→      // Corrupt draft — ignore, start fresh.
   437→    }
   438→  }, []);
   439→
   440→  // Persist the draft on every keystroke; cleared once the form is empty.
   441→  // Only content fields count — the name alone lives in its own key.
   442→  useEffect(() => {
   443→    const draft: FormDraft = { actorName, title, durationHours, notes };
   444→    const hasContent = Boolean(
   445→      title.trim() || durationHours.trim() || notes.trim()
   446→    );
   447→    if (hasContent) {
   448→      window.localStorage.setItem(DRAFT_STORAGE_KEY, JSON.stringify(draft));
   449→    } else {
   450→      window.localStorage.removeItem(DRAFT_STORAGE_KEY);
   451→    }
   452→  }, [actorName, title, durationHours, notes]);
   453→
   454→  const isToday = !dateValue || (todayKey !== "" && dateValue === todayKey);
   455→  const isYesterday =
   456→    !isToday &&
   457→    todayKey !== "" &&
   458→    dateValue === toDateKey(new Date(Date.now() - 86400000));
   459→
   460→  // "Use last practice" is only useful while the form is still blank.
   461→  const canReuse = Boolean(lastLog) && !title && !durationHours && !notes;
   462→  const reuseLast = () => {
   463→    if (!lastLog) return;
   464→    setTitle(lastLog.title);
   465→    setDurationHours(minutesToHoursInput(lastLog.durationMin));
   466→    setNotes(lastLog.notes);
   467→  };
   468→
   469→  const pickedDateLabel =
   470→    dateValue && !isToday
   471→      ? new Date(`${dateValue}T00:00:00`).toLocaleDateString("id-ID", {
   472→          day: "numeric",
   473→          month: "long",
   474→        })
   475→      : "hari ini";
   476→
   477→  const pickPhoto = (file: File | null) => {
   478→    if (photoPreview) URL.revokeObjectURL(photoPreview);
   479→    setPhoto(file);
   480→    setPhotoPreview(file ? URL.createObjectURL(file) : null);
   481→  };
   482→
   483→  const reset = (keepName: boolean) => {
   484→    if (!keepName) setActorName("");
   485→    setTitle("");
   486→    setDurationHours("");
   487→    setNotes("");
   488→    setDateValue("");
   489→    setShowDatePicker(false);
   490→    setDraftRestored(false);
   491→    pickPhoto(null);
   492→    if (fileRef.current) fileRef.current.value = "";
   493→  };
   494→
   495→  const submit = async (e?: React.FormEvent) => {
   496→    e?.preventDefault();
   497→    setFieldError(null);
   498→
   499→    if (!actorName.trim()) return setFieldError("Isi nama/divisi dulu ya.");
   500→    if (!title.trim()) return setFieldError("Tulis apa yang dilatih hari ini.");
   501→    const minutes = hoursToMinutes(durationHours);
   502→    if (minutes === null || minutes < 1) {
   503→      return setFieldError("Isi durasi latihan (jam).");
   504→    }
   505→    if (minutes > 1440) {
   506→      return setFieldError("Durasi maksimal 24 jam.");
   507→    }
   508→
   509→    setSubmitting(true);
   510→    try {
   511→      const fd = new FormData();
   512→      fd.set("actorName", actorName.trim());
   513→      fd.set("title", title.trim());
   514→      fd.set("durationMin", String(minutes));
   515→      fd.set("notes", notes.trim());
   516→      if (!isToday && dateValue) fd.set("date", dateValue);
   517→
   518→      let file: File | null = photo;
   519→      if (file) {
   520→        // Downscale in the browser so uploads stay small.
   521→        file = await downscaleImage(file);
   522→        if (file) fd.set("image", file);
   523→      }
   524→
   525→      const res = await fetch("/api/logs", { method: "POST", body: fd });
   526→      const data = (await res.json().catch(() => null)) as
   527→        | { log?: PracticeLog; error?: string }
   528→        | null;
   529→
   530→      if (!res.ok || !data?.log) {
   531→        throw new Error(data?.error ?? "Gagal menyimpan. Coba lagi.");
   532→      }
   533→
   534→      window.localStorage.setItem(ACTOR_NAME_STORAGE_KEY, actorName.trim());
   535→      onCreated(data.log);
   536→      reset(true);
   537→      toast.success("Latihan tercatat.");
   538→    } catch (err) {
   539→      setFieldError(

   788→function relativeDayLabel(day: string): string | null {
   789→  const today = toDateKey(new Date());
   790→  if (day === today) return "Hari ini";
   791→  const target = new Date(`${day}T00:00:00`).getTime();
   792→  const now = new Date(`${today}T00:00:00`).getTime();
   793→  const diffDays = Math.round((now - target) / 86400000);
   794→  if (diffDays === 1) return "Kemarin";
   795→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   796→  return null;
   797→}

   801→  justAdded,
   802→  onDeleted,
   803→  onUpdated,
   804→}: {
   805→  log: PracticeLog;
   806→  justAdded: boolean;
   807→  onDeleted: (id: string) => void;
   808→  onUpdated: (log: PracticeLog) => void;

   810→  const [confirming, setConfirming] = useState(false);
   811→  const [editing, setEditing] = useState(false);
   812→  const [saving, setSaving] = useState(false);
   813→  const [editError, setEditError] = useState<string | null>(null);
   814→  const [focusAfterEdit, setFocusAfterEdit] = useState(false);
   815→  const editButtonRef = useRef<HTMLButtonElement>(null);
   816→  const [draft, setDraft] = useState({
   817→    actorName: log.actorName,
   818→    title: log.title,
   819→    durationHours: minutesToHoursInput(log.durationMin),

   825→    setDraft({

  1071→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
  1072→async function downscaleImage(file: File): Promise<File | null> {
  1073→  try {
  1074→    const bitmap = await createImageBitmap(file);
  1075→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  1076→    const w = Math.round(bitmap.width * scale);
  1077→    const h = Math.round(bitmap.height * scale);
  1078→
  1079→    const canvas = document.createElement("canvas");
  1080→    canvas.width = w;
  1081→    canvas.height = h;
  1082→    const ctx = canvas.getContext("2d");
  1083→    if (!ctx) return file;
  1084→    ctx.drawImage(bitmap, 0, 0, w, h);
  1085→    bitmap.close();
  1086→
  1087→    const blob = await new Promise<Blob | null>((resolve) =>
  1088→      canvas.toBlob(resolve, "image/jpeg", 0.85)
  1089→    );
  1090→    if (!blob) return file;
  1091→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
  1092→      type: "image/jpeg",
  1093→    });
  1094→  } catch {
  1095→    return file; // fall back to the original file
  1096→  }
  1097→}
