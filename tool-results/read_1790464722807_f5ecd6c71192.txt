   397→   397→function LogForm({
   398→   398→  onCreated,
   399→   399→  lastLog,
   400→   400→}: {
   401→   401→  onCreated: (log: PracticeLog) => void;
   402→   402→  lastLog: PracticeLog | null;
   403→   403→}) {
   404→   404→  const [actorName, setActorName] = useState("");
   405→   405→  const [title, setTitle] = useState("");
   406→   406→  const [durationHours, setDurationHours] = useState("");
   407→   407→  const [notes, setNotes] = useState("");
   408→   408→  const [dateValue, setDateValue] = useState(""); // "" = today
   409→   409→  const [showDatePicker, setShowDatePicker] = useState(false);
   410→   410→  const [todayKey, setTodayKey] = useState("");
   411→   411→  const [photo, setPhoto] = useState<File | null>(null);
   412→   412→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   413→   413→  const [submitting, setSubmitting] = useState(false);
   414→   414→  const [fieldError, setFieldError] = useState<string | null>(null);
   415→   415→  const [draftRestored, setDraftRestored] = useState(false);
   416→   416→  const fileRef = useRef<HTMLInputElement>(null);
   417→   417→
   418→   418→  // Prefill name from localStorage (client only), then restore any unsent
   419→   419→  // draft so a reload never loses what was being typed.
   420→   420→  useEffect(() => {
   421→   421→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   422→   422→    if (saved) setActorName(saved);
   423→   423→    setTodayKey(toDateKey(new Date()));
   424→   424→
   425→   425→    try {
   426→   426→      const raw = window.localStorage.getItem(DRAFT_STORAGE_KEY);
   427→   427→      if (!raw) return;
   428→   428→      const d = JSON.parse(raw) as Partial<FormDraft>;
   429→   429→      if (!d.title && !d.durationHours && !d.notes) return;
   430→   430→      if (d.actorName) setActorName(d.actorName);
   431→   431→      if (d.title) setTitle(d.title);
   432→   432→      if (d.durationHours) setDurationHours(d.durationHours);
   433→   433→      if (d.notes) setNotes(d.notes);
   434→   434→      setDraftRestored(true);
   435→   435→    } catch {
   436→   436→      // Corrupt draft — ignore, start fresh.
   437→   437→    }
   438→   438→  }, []);
   439→   439→
   440→   440→  // Persist the draft on every keystroke; cleared once the form is empty.
   441→   441→  // Only content fields count — the name alone lives in its own key.
   442→   442→  useEffect(() => {
   443→   443→    const draft: FormDraft = { actorName, title, durationHours, notes };
   444→   444→    const hasContent = Boolean(
   445→   445→      title.trim() || durationHours.trim() || notes.trim()

   788→   788→function relativeDayLabel(day: string): string | null {
   789→   789→  const today = toDateKey(new Date());
   790→   790→  if (day === today) return "Hari ini";
   791→   791→  const target = new Date(`${day}T00:00:00`).getTime();
   792→   792→  const now = new Date(`${today}T00:00:00`).getTime();
   793→   793→  const diffDays = Math.round((now - target) / 86400000);
   794→   794→  if (diffDays === 1) return "Kemarin";
   795→   795→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   796→   796→  return null;

   799→   799→function LogRow({
   800→   800→  log,
   801→   801→  justAdded,
   802→   802→  onDeleted,
   803→   803→  onUpdated,
   804→   804→}: {
   805→   805→  log: PracticeLog;
   806→   806→  justAdded: boolean;
   807→   807→  onDeleted: (id: string) => void;
   808→   808→  onUpdated: (log: PracticeLog) => void;
   809→   809→}) {
   810→   810→  const [confirming, setConfirming] = useState(false);
   811→   811→  const [editing, setEditing] = useState(false);
   812→   812→  const [saving, setSaving] = useState(false);
   813→   813→  const [editError, setEditError] = useState<string | null>(null);
   814→   814→  const [focusAfterEdit, setFocusAfterEdit] = useState(false);
   815→   815→  const editButtonRef = useRef<HTMLButtonElement>(null);
   816→   816→  const [draft, setDraft] = useState({
   817→   817→    actorName: log.actorName,
   818→   818→    title: log.title,
   819→   819→    durationHours: minutesToHoursInput(log.durationMin),
   820→   820→    notes: log.notes,
   821→   821→    date: dateKey(log.date),
   822→   822→  });
   823→   823→
   824→   824→  const startEdit = () => {
   825→   825→    setDraft({
   826→   826→      actorName: log.actorName,
   827→   827→      title: log.title,
   828→   828→      durationHours: minutesToHoursInput(log.durationMin),
   829→   829→      notes: log.notes,
   830→   830→      date: dateKey(log.date),
   831→   831→    });
   832→   832→    setEditError(null);
   833→   833→    setConfirming(false);
   834→   834→    setEditing(true);
   835→   835→  };
   836→   836→
   837→   837→  /** Leave edit mode and hand focus back to the "ubah" link once it re-mounts. */
   838→   838→  const cancelEdit = () => {
   839→   839→    setEditing(false);
   840→   840→    setFocusAfterEdit(true);
   841→   841→  };
   842→   842→
   843→   843→  // Focus is restored in a post-commit effect so the ref'd "ubah" button
   844→   844→  // is guaranteed to exist (works for both cancel and save paths).
   845→   845→  useEffect(() => {
   846→   846→    if (!editing && focusAfterEdit) {
   847→   847→      setFocusAfterEdit(false);
   848→   848→      editButtonRef.current?.focus();
   849→   849→    }
   850→   850→  }, [editing, focusAfterEdit]);
   851→   851→
   852→   852→  const saveEdit = async () => {
   853→   853→    const name = draft.actorName.trim();
   854→   854→    const title = draft.title.trim();
   855→   855→    const minutes = hoursToMinutes(draft.durationHours);
   856→   856→    if (!name) return setEditError("Nama tidak boleh kosong.");
   857→   857→    if (!title) return setEditError("Judul tidak boleh kosong.");
   858→   858→    if (!draft.date) return setEditError("Tanggal wajib ada.");
   859→   859→    if (minutes === null || minutes < 1) {
   860→   860→      return setEditError("Isi durasi (jam).");
   861→   861→    }
   862→   862→    if (minutes > 1440) {
   863→   863→      return setEditError("Durasi maksimal 24 jam.");
   864→   864→    }
   865→   865→
   866→   866→    setSaving(true);
   867→   867→    setEditError(null);
   868→   868→    try {
   869→   869→      const dateChanged = draft.date !== dateKey(log.date);
   870→   870→      const res = await fetch(`/api/logs/${log.id}`, {
   871→   871→        method: "PATCH",
   872→   872→        headers: { "Content-Type": "application/json" },
   873→   873→        body: JSON.stringify({
   874→   874→          actorName: name,
   875→   875→          title,
   876→   876→          durationMin: minutes,
   877→   877→          notes: draft.notes.trim(),
   878→   878→          ...(dateChanged ? { date: draft.date } : {}),
   879→   879→        }),
   880→   880→      });
   881→   881→      const data = (await res.json().catch(() => null)) as
   882→   882→        | { log?: PracticeLog; error?: string }
   883→   883→        | null;
   884→   884→      if (!res.ok || !data?.log) {
   885→   885→        throw new Error(data?.error ?? "Gagal menyimpan perubahan.");
   886→   886→      }
   887→   887→      onUpdated(data.log);
   888→   888→      setEditing(false);
   889→   889→      setFocusAfterEdit(true);
   890→   890→      toast.success("Perubahan tersimpan.");
   891→   891→    } catch (err) {
   892→   892→      setEditError(err instanceof Error ? err.message : "Gagal menyimpan perubahan.");
   893→   893→    } finally {
   894→   894→      setSaving(false);
   895→   895→    }
   896→   896→  };
   897→   897→
   898→   898→  const remove = async () => {
   899→   899→    if (!confirming) {
   900→   900→      setConfirming(true);
   901→   901→      return;
   902→   902→    }
   903→   903→    try {
   904→   904→      const res = await fetch(`/api/logs/${log.id}`, { method: "DELETE" });
   905→   905→      if (!res.ok) throw new Error();
   906→   906→      onDeleted(log.id);
   907→   907→      toast.success("Catatan dihapus.");
   908→   908→    } catch {
   909→   909→      toast.error("Gagal menghapus. Coba lagi.");
   910→   910→      setConfirming(false);
   911→   911→    }
   912→   912→  };
   913→   913→
   914→   914→  const day = dateKey(log.date);
   915→   915→  const dayLabel =
   916→   916→    relativeDayLabel(day) ??

  1071→  1071→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
  1072→  1072→async function downscaleImage(file: File): Promise<File | null> {
  1073→  1073→  try {
  1074→  1074→    const bitmap = await createImageBitmap(file);
  1075→  1075→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  1076→  1076→    const w = Math.round(bitmap.width * scale);
  1077→  1077→    const h = Math.round(bitmap.height * scale);
  1078→  1078→
  1079→  1079→    const canvas = document.createElement("canvas");
  1080→  1080→    canvas.width = w;
  1081→  1081→    canvas.height = h;
  1082→  1082→    const ctx = canvas.getContext("2d");
  1083→  1083→    if (!ctx) return file;
  1084→  1084→    ctx.drawImage(bitmap, 0, 0, w, h);
  1085→  1085→    bitmap.close();
  1086→  1086→
  1087→  1087→    const blob = await new Promise<Blob | null>((resolve) =>
  1088→  1088→      canvas.toBlob(resolve, "image/jpeg", 0.85)
  1089→  1089→    );
  1090→  1090→    if (!blob) return file;
  1091→  1091→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
  1092→  1092→      type: "image/jpeg",
  1093→  1093→    });
  1094→  1094→  } catch {
  1095→  1095→    return file; // fall back to the original file
  1096→  1096→  }
  1097→  1097→}
