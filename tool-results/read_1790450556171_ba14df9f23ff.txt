    40→      const res = await fetch("/api/logs", { cache: "no-store" });
    41→      if (!res.ok) throw new Error();

    43→      setLogs(data.logs);
    44→    } catch {
    45→      setError("Tidak bisa memuat log. Coba lagi.");
    46→    } finally {
    47→      setLoading(false);
    48→    }
    49→  }, []);
    50→
    51→  useEffect(() => {
    52→    void refresh();
    53→  }, [refresh]);
    54→
    55→  const handleCreated = useCallback((log: PracticeLog) => {
    56→    setLogs((prev) => {
    57→      // Insert in feed order (date desc, createdAt desc) so backdated
    58→      // entries land in the right place instead of always on top.
    59→      const next = [...prev, log];
    60→      next.sort((a, b) => {
    61→        const byDate = b.date.localeCompare(a.date);
    62→        if (byDate !== 0) return byDate;
    63→        return b.createdAt.localeCompare(a.createdAt);
    64→      });
    65→      return next;
    66→    });
    67→    setJustAddedId(log.id);
    68→    window.setTimeout(() => {
    69→      setJustAddedId((current) => (current === log.id ? null : current));
    70→      document
    71→        .getElementById(`log-${log.id}`)
    72→        ?.scrollIntoView({ behavior: "smooth", block: "nearest" });
    73→    }, 120);
    74→  }, []);
    75→
    76→  const handleDeleted = useCallback((id: string) => {
    77→    setLogs((prev) => prev.filter((l) => l.id !== id));
    78→  }, []);
    79→
    80→  const handleUpdated = useCallback((updated: PracticeLog) => {
    81→    setLogs((prev) => prev.map((l) => (l.id === updated.id ? updated : l)));
    82→  }, []);
    83→
    84→  const totalMin = logs.reduce((sum, l) => sum + l.durationMin, 0);
    85→  const groups = groupByMonthWeek(logs);
    86→
    87→  return (

    89→      {/* washi paper grain */}
    90→      <div aria-hidden className="washi-grain pointer-events-none fixed inset-0 -z-10" />
    91→      {/* vertical tategaki accent, wide screens only */}
    92→      <p
    93→        aria-hidden
    94→        className="font-kanji fixed right-6 top-1/2 hidden -translate-y-1/2 select-none text-sm tracking-[0.6em] text-foreground/15 lg:block [writing-mode:vertical-rl]"
    95→      >
    96→        稽古日誌
    97→      </p>
    98→
    99→      {/* Header */}
   100→      <header className="border-b border-foreground/10">
   101→        <div className="mx-auto flex h-16 w-full max-w-2xl items-center justify-between px-5">
   102→          <div className="flex items-center gap-3">
   103→            <span
   104→              aria-hidden
   105→              className="font-kanji flex size-9 flex-col items-center justify-center rounded-[3px] bg-seal text-[11px] leading-[1.2] text-[#f7f2e6] ring-1 ring-inset ring-white/25"
   106→            >
   107→              <span>稽</span>
   108→              <span>古</span>
   109→            </span>
   110→            <div>
   111→              <p className="font-serif text-sm leading-tight tracking-wide">稽古日誌</p>
   112→              <p className="text-[10px] uppercase tracking-[0.3em] text-muted-foreground">
   113→                Log Latihan Teater
   114→              </p>
   115→            </div>
   116→          </div>
   117→          <p className="text-xs tabular-nums text-muted-foreground">
   118→            {new Date().toLocaleDateString("id-ID", {
   119→              weekday: "long",
   120→              day: "numeric",
   121→              month: "long",
   122→            })}
   123→          </p>
   124→        </div>
   125→      </header>
   126→
   127→      <main className="mx-auto w-full max-w-2xl flex-1 px-5">
   128→        {/* Form */}
   129→        <LogForm onCreated={handleCreated} lastLog={logs[0] ?? null} />
   130→
   131→        {/* Summary */}
   132→        {!loading && !error && logs.length > 0 && (
   133→          <p aria-live="polite" className="mb-6 text-sm tabular-nums text-muted-foreground">
   134→            {logs.length} sesi · {formatDuration(totalMin)} total
   135→          </p>
   136→        )}
   137→
   138→        {/* Feed */}
   139→        {loading ? (
   140→          <div className="space-y-3 pb-16">
   141→            <div className="h-5 w-40 rounded bg-foreground/5" />
   142→            <div className="h-24 rounded-[3px] border border-foreground/10" />
   143→            <div className="h-24 rounded-[3px] border border-foreground/10" />
   144→          </div>
   145→        ) : error ? (
   146→          <div className="pb-16 text-sm text-muted-foreground">
   147→            <p>{error}</p>
   148→            <Button variant="outline" size="sm" className="mt-3" onClick={() => void refresh()}>
   149→              Coba lagi

   151→          </div>

   166→                month.weeks.flatMap((w) =>
   167→                  w.logs.map((l) => l.actorName.trim().toLowerCase())
   168→                )
   169→              ).size;
   170→              return (
   171→                <section key={month.key}>
   172→                  {/* thick-thin double rule, Showa print style */}
   173→                  <div className="flex items-baseline justify-between gap-3 border-b-[3px] border-double border-foreground/25 pb-2">
   174→                    <h2 className="font-serif text-base tracking-wide">
   175→                      {month.label}
   176→                      <span aria-hidden className="font-kanji ml-2 text-xs text-seal">
   177→                        {Number(month.key.split("-")[1])}月

   181→                        {actorCount > 1 ? ` · ${actorCount} orang` : ""}

   200→                          <LogRow
   201→                            key={log.id}
   202→                            log={log}
   203→                            justAdded={log.id === justAddedId}
   204→                            onDeleted={handleDeleted}
   205→                            onUpdated={handleUpdated}
   206→                          />
   207→                        ))}
   208→                      </ul>
   209→                    </div>

   214→          </div>
   215→        )}
   216→      </main>
   217→
   218→      {/* Footer */}
   219→      <footer className="mt-auto border-t border-foreground/10">

   222→            <p className="font-kanji text-sm tracking-[0.3em] text-foreground/70">継続は力なり</p>
   223→            <p className="mt-1.5 text-xs text-muted-foreground">
   224→              Keizoku wa chikara nari — konsistensi adalah kekuatan.
   225→            </p>
   226→          </div>
   227→          <button
   228→            type="button"
   229→            onClick={() =>
   230→              window.scrollTo({
   231→                top: 0,
   232→                behavior: window.matchMedia("(prefers-reduced-motion: reduce)").matches
   233→                  ? "auto"
   234→                  : "smooth",
   235→              })
   236→            }
   237→            className={`${textLink} shrink-0 pt-0.5`}
   238→          >
   239→            ↑ atas
   240→          </button>
   241→        </div>
   242→      </footer>
   243→    </div>
   244→  );
   245→}

   249→/** Quiet link that opens WhatsApp with the month recap pre-filled. */
   250→function ShareMonthLink({ month }: { month: MonthGroup }) {
   251→  const href = `https://wa.me/?text=${encodeURIComponent(buildMonthSummary(month))}`;
   252→  return (
   253→    <a
   254→      href={href}
   255→      target="_blank"
   256→      rel="noreferrer"
   257→      title="Bagikan ringkasan bulan ini via WhatsApp"
   258→      className={textLink}
   259→    >
   260→      whatsapp
   261→    </a>
   262→  );
   263→}
   264→
   265→function CopyMonthButton({ month }: { month: MonthGroup }) {
   266→  const [copied, setCopied] = useState(false);
   267→
   268→  const copy = async () => {
   269→    try {
   270→      await navigator.clipboard.writeText(buildMonthSummary(month));
   271→      toast.success("Ringkasan disalin.");
   272→      setCopied(true);
   273→      window.setTimeout(() => setCopied(false), 1600);
   274→    } catch {
   275→      toast.error("Gagal menyalin.");
   276→    }
   277→  };
   278→
   279→  return (
   280→    <button type="button" onClick={copy} className={textLink}>
   281→      {copied ? "tersalin" : "salin"}
   282→    </button>
   283→  );
   284→}

   296→function LogForm({
   297→  onCreated,
   298→  lastLog,
   299→}: {
   300→  onCreated: (log: PracticeLog) => void;
   301→  lastLog: PracticeLog | null;
   302→}) {
   303→  const [actorName, setActorName] = useState("");
   304→  const [title, setTitle] = useState("");
   305→  const [durationHours, setDurationHours] = useState("");
   306→  const [notes, setNotes] = useState("");
   307→  const [dateValue, setDateValue] = useState(""); // "" = today
   308→  const [showDatePicker, setShowDatePicker] = useState(false);
   309→  const [todayKey, setTodayKey] = useState("");
   310→  const [photo, setPhoto] = useState<File | null>(null);
   311→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   312→  const [submitting, setSubmitting] = useState(false);
   313→  const [fieldError, setFieldError] = useState<string | null>(null);
   314→  const fileRef = useRef<HTMLInputElement>(null);
   315→
   316→  // Prefill name from localStorage (client only).
   317→  useEffect(() => {
   318→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   319→    if (saved) setActorName(saved);
   320→    setTodayKey(toDateKey(new Date()));
   321→  }, []);
   322→
   323→  const isToday = !dateValue || (todayKey !== "" && dateValue === todayKey);
   324→  const isYesterday =
   325→    !isToday &&
   326→    todayKey !== "" &&
   327→    dateValue === toDateKey(new Date(Date.now() - 86400000));
   328→
   329→  // "Use last practice" is only useful while the form is still blank.
   330→  const canReuse = Boolean(lastLog) && !title && !durationHours && !notes;
   331→  const reuseLast = () => {
   332→    if (!lastLog) return;
   333→    setTitle(lastLog.title);
   334→    setDurationHours(minutesToHoursInput(lastLog.durationMin));
   335→    setNotes(lastLog.notes);
   336→  };
   337→
   338→  const pickedDateLabel =
   339→    dateValue && !isToday
   340→      ? new Date(`${dateValue}T00:00:00`).toLocaleDateString("id-ID", {
   341→          day: "numeric",
   342→          month: "long",
   343→        })
   344→      : "hari ini";
   345→
   346→  const pickPhoto = (file: File | null) => {
   347→    if (photoPreview) URL.revokeObjectURL(photoPreview);
   348→    setPhoto(file);
   349→    setPhotoPreview(file ? URL.createObjectURL(file) : null);
   350→  };
   351→
   352→  const reset = (keepName: boolean) => {
   353→    if (!keepName) setActorName("");
   354→    setTitle("");
   355→    setDurationHours("");
   356→    setNotes("");
   357→    setDateValue("");
   358→    setShowDatePicker(false);
   359→    pickPhoto(null);
   360→    if (fileRef.current) fileRef.current.value = "";
   361→  };
   362→
   363→  const submit = async (e?: React.FormEvent) => {
   364→    e?.preventDefault();
   365→    setFieldError(null);
   366→
   367→    if (!actorName.trim()) return setFieldError("Isi nama/divisi dulu ya.");
   368→    if (!title.trim()) return setFieldError("Tulis apa yang dilatih hari ini.");
   369→    const minutes = hoursToMinutes(durationHours);
   370→    if (minutes === null || minutes < 1) {
   371→      return setFieldError("Isi durasi latihan (jam).");
   372→    }
   373→    if (minutes > 1440) {
   374→      return setFieldError("Durasi maksimal 24 jam.");
   375→    }
   376→
   377→    setSubmitting(true);
   378→    try {
   379→      const fd = new FormData();
   380→      fd.set("actorName", actorName.trim());
   381→      fd.set("title", title.trim());
   382→      fd.set("durationMin", String(minutes));
   383→      fd.set("notes", notes.trim());
   384→      if (!isToday && dateValue) fd.set("date", dateValue);
   385→
   386→      let file: File | null = photo;
   387→      if (file) {
   388→        // Downscale in the browser so uploads stay small.
   389→        file = await downscaleImage(file);
   390→        if (file) fd.set("image", file);
   391→      }
   392→
   393→      const res = await fetch("/api/logs", { method: "POST", body: fd });
   394→      const data = (await res.json().catch(() => null)) as
   395→        | { log?: PracticeLog; error?: string }
   396→        | null;
   397→
   398→      if (!res.ok || !data?.log) {
   399→        throw new Error(data?.error ?? "Gagal menyimpan. Coba lagi.");
   400→      }
   401→
   402→      window.localStorage.setItem(ACTOR_NAME_STORAGE_KEY, actorName.trim());
   403→      onCreated(data.log);
   404→      reset(true);
   405→      toast.success("Latihan tercatat.");
   406→    } catch (err) {
   407→      setFieldError(err instanceof Error ? err.message : "Gagal menyimpan. Coba lagi.");
   408→    } finally {
   409→      setSubmitting(false);
   410→    }
   411→  };
   412→
   413→  return (
   414→    <form
   415→      onSubmit={submit}
   416→      onKeyDown={(e) => {
   417→        if (e.key === "Enter" && (e.metaKey || e.ctrlKey)) {
   418→          e.preventDefault();
   419→          void submit();
   420→        }
   421→      }}
   422→      className="pb-8 pt-10"
   423→      noValidate
   424→    >
   425→      <p
   426→        aria-hidden
   427→        className="font-kanji text-xs tracking-[0.35em] text-seal"
   428→      >
   429→        {isToday ? "今日の稽古" : isYesterday ? "昨日の稽古" : "過去の稽古"}
   430→      </p>
   431→      <h1 className="mt-2 font-serif text-2xl tracking-tight">
   432→        {isToday ? "Catat latihan hari ini" : `Catat latihan ${pickedDateLabel}`}
   433→      </h1>
   434→      <p className="mt-2 text-sm text-muted-foreground">
   435→        Isi detilnya, tambahkan foto kalau ada, lalu kirim.
   436→      </p>
   437→      {canReuse && (
   438→        <p className="mt-2 text-xs text-muted-foreground">
   439→          Latihan terakhir: “{lastLog?.title}” ·{" "}
   440→          <button type="button" onClick={reuseLast} className={textLink}>
   441→            pakai lagi
   442→          </button>
   443→        </p>
   444→      )}
   445→
   446→      {/* Date row — quiet by default, backdate on demand */}
   447→      <div className="mt-4 min-h-9">
   448→        {showDatePicker ? (
   449→          <div className="flex flex-wrap items-center gap-3">
   450→            <input
   451→              type="date"
   452→              value={dateValue}
   453→              max={todayKey || undefined}
   454→              onChange={(e) => setDateValue(e.target.value)}
   455→              aria-label="Tanggal latihan"
   456→              className="h-9 rounded-[3px] border border-foreground/15 bg-card/70 px-2 text-sm tabular-nums shadow-none outline-none transition-colors focus:border-seal"
   457→            />
   458→            <button
   459→              type="button"
   460→              onClick={() => setShowDatePicker(false)}
   461→              className={textLink}
   462→            >
   463→              selesai
   464→            </button>
   465→          </div>
   466→        ) : (
   467→          <p className="text-xs text-muted-foreground">
   468→            Tanggal: {pickedDateLabel} ·{" "}
   469→            <button
   470→              type="button"
   471→              onClick={() => setShowDatePicker(true)}
   472→              className={textLink}
   473→            >
   474→              ubah
   475→            </button>
   476→          </p>
   477→        )}
   478→      </div>
   479→
   480→      <div className="mt-6 space-y-5">
   481→        <div className="grid gap-5 sm:grid-cols-2">
   482→          <div className="space-y-1.5">
   483→            <Label htmlFor="actorName">Nama/Divisi</Label>
   484→            <Input
   485→              id="actorName"
   486→              value={actorName}
   487→              onChange={(e) => setActorName(e.target.value)}
   488→              placeholder="cth. Raka — Divisi Akting"
   489→              autoComplete="name"
   490→              className={underlineInput}
   491→            />
   492→          </div>
   493→          <div className="space-y-1.5">
   494→            <Label htmlFor="durationHours">Durasi (jam)</Label>
   495→            <Input
   496→              id="durationHours"
   497→              type="text"
   498→              inputMode="decimal"
   499→              autoComplete="off"
   500→              value={durationHours}
   501→              onChange={(e) => setDurationHours(e.target.value)}
   502→              placeholder="1,5"
   503→              className={`${underlineInput} tabular-nums`}
   504→            />
   505→            <div className="flex items-center gap-1">
   506→              {DURATION_CHIPS.map((chip) => (

   544→          />

   546→

   605→function relativeDayLabel(day: string): string | null {
   606→  const today = toDateKey(new Date());
   607→  if (day === today) return "Hari ini";
   608→  const target = new Date(`${day}T00:00:00`).getTime();
   609→  const now = new Date(`${today}T00:00:00`).getTime();
   610→  const diffDays = Math.round((now - target) / 86400000);
   611→  if (diffDays === 1) return "Kemarin";
   612→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   613→  return null;
   614→}

   616→function LogRow({
   617→  log,
   618→  justAdded,
   619→  onDeleted,
   620→  onUpdated,
   621→}: {
   622→  log: PracticeLog;
   623→  justAdded: boolean;
   624→  onDeleted: (id: string) => void;
   625→  onUpdated: (log: PracticeLog) => void;
   626→}) {
   627→  const [confirming, setConfirming] = useState(false);
   628→  const [editing, setEditing] = useState(false);
   629→  const [saving, setSaving] = useState(false);
   630→  const [editError, setEditError] = useState<string | null>(null);
   631→  const [draft, setDraft] = useState({
   632→    actorName: log.actorName,
   633→    title: log.title,
   634→    durationHours: minutesToHoursInput(log.durationMin),
   635→    notes: log.notes,
   636→  });
   637→
   638→  const startEdit = () => {
   639→    setDraft({
   640→      actorName: log.actorName,
   641→      title: log.title,
   642→      durationHours: minutesToHoursInput(log.durationMin),
   643→      notes: log.notes,
   644→    });
   645→    setEditError(null);
   646→    setConfirming(false);
   647→    setEditing(true);
   648→  };
   649→
   650→  const saveEdit = async () => {
   651→    const name = draft.actorName.trim();
   652→    const title = draft.title.trim();
   653→    const minutes = hoursToMinutes(draft.durationHours);
   654→    if (!name) return setEditError("Nama tidak boleh kosong.");
   655→    if (!title) return setEditError("Judul tidak boleh kosong.");
   656→    if (minutes === null || minutes < 1) {
   657→      return setEditError("Isi durasi (jam).");
   658→    }
   659→    if (minutes > 1440) {
   660→      return setEditError("Durasi maksimal 24 jam.");
   661→    }
   662→
   663→    setSaving(true);
   664→    setEditError(null);
   665→    try {
   666→      const res = await fetch(`/api/logs/${log.id}`, {
   667→        method: "PATCH",
   668→        headers: { "Content-Type": "application/json" },
   669→        body: JSON.stringify({
   670→          actorName: name,
   671→          title,
   672→          durationMin: minutes,
   673→          notes: draft.notes.trim(),
   674→        }),
   675→      });
   676→      const data = (await res.json().catch(() => null)) as
   677→        | { log?: PracticeLog; error?: string }
   678→        | null;
   679→      if (!res.ok || !data?.log) {
   680→        throw new Error(data?.error ?? "Gagal menyimpan perubahan.");
   681→      }
   682→      onUpdated(data.log);
   683→      setEditing(false);
   684→      toast.success("Perubahan tersimpan.");
   685→    } catch (err) {
   686→      setEditError(err instanceof Error ? err.message : "Gagal menyimpan perubahan.");
   687→    } finally {
   688→      setSaving(false);
   689→    }
   690→  };
   691→
   692→  const remove = async () => {
   693→    if (!confirming) {
   694→      setConfirming(true);
   695→      return;
   696→    }
   697→    try {
   698→      const res = await fetch(`/api/logs/${log.id}`, { method: "DELETE" });
   699→      if (!res.ok) throw new Error();
   700→      onDeleted(log.id);
   701→      toast.success("Catatan dihapus.");
   702→    } catch {
   703→      toast.error("Gagal menghapus. Coba lagi.");
   704→      setConfirming(false);
   705→    }
   706→  };
   707→
   708→  const day = dateKey(log.date);
   709→  const dayLabel =
   710→    relativeDayLabel(day) ??
   711→    new Date(`${day}T00:00:00`).toLocaleDateString("id-ID", {
   712→      day: "numeric",
   713→      month: "short",
   714→    });
   715→
   716→  if (editing) {
   717→    return (
   718→      <li className="rounded-[3px] border border-foreground/30 bg-card p-4">
   719→        <div className="space-y-4">
   720→          <div className="grid gap-4 sm:grid-cols-2">
   721→            <input
   722→              value={draft.actorName}
   723→              onChange={(e) => setDraft({ ...draft, actorName: e.target.value })}
   724→              aria-label="Nama/Divisi"
   725→              className={underlineInput}
   726→              autoFocus
   727→            />
   728→            <input
   729→              value={draft.durationHours}
   730→              onChange={(e) => setDraft({ ...draft, durationHours: e.target.value })}
   731→              type="text"
   732→              inputMode="decimal"
   733→              autoComplete="off"
   734→              aria-label="Durasi (jam)"
   735→              className={`${underlineInput} tabular-nums`}
   736→            />
   737→          </div>
   738→          <input
   739→            value={draft.title}
   740→            onChange={(e) => setDraft({ ...draft, title: e.target.value })}
   741→            aria-label="Apa yang dilatih"
   742→            className={underlineInput}
   743→          />
   744→          <textarea
   745→            value={draft.notes}
   746→            onChange={(e) => setDraft({ ...draft, notes: e.target.value })}
   747→            rows={3}
   748→            aria-label="Catatan"
   749→            className="w-full resize-y rounded-[3px] border border-foreground/15 bg-transparent px-3 py-2 text-sm outline-none transition-colors focus:border-seal"
   750→          />
   751→          {editError && (
   752→            <p role="alert" className="text-sm text-destructive">
   753→              {editError}
   754→            </p>
   755→          )}
   756→          <div className="flex items-center justify-end gap-4">
   757→            <button type="button" onClick={() => setEditing(false)} className={textLink}>
   758→              batal
   759→            </button>
   760→            <button
   761→              type="button"
   762→              onClick={saveEdit}
   763→              disabled={saving}
   764→              className="rounded-sm text-xs font-semibold underline underline-offset-2 disabled:opacity-50 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring"
   765→            >
   766→              {saving ? "menyimpan…" : "simpan"}
   767→            </button>
   768→          </div>
   769→        </div>
   770→      </li>
   771→    );
   772→  }
   773→
   774→  return (
   775→    <li
   776→      id={`log-${log.id}`}
   777→      className={`rounded-[3px] border bg-card p-4 transition-colors hover:border-foreground/30 ${
   778→        justAdded ? "entry-flash border-seal/40" : "border-foreground/10"
   779→      }`}
   780→    >
   781→      <div className="flex items-baseline justify-between gap-3">
   782→        <p className="font-serif text-sm font-bold">{log.actorName}</p>
   783→        <p className="shrink-0 text-xs tabular-nums text-muted-foreground">
   784→          {dayLabel} · {formatDuration(log.durationMin)}
   785→        </p>
   786→      </div>
   787→      <p className="mt-1 text-sm">{log.title}</p>
   788→      {log.notes && (
   789→        <p className="mt-1.5 whitespace-pre-line text-sm text-muted-foreground">{log.notes}</p>
   790→      )}
   791→      {log.imagePath && (
   792→        <a
   793→          href={log.imagePath}
   794→          target="_blank"
   795→          rel="noreferrer"
   796→          className={`group mt-4 block ${
   797→            // Prints taped into a Showa photo album: a hairline white frame,
   798→            // the faintest tilt, straightening on hover.
   799→            log.id.charCodeAt(0) % 2 === 0 ? "-rotate-[0.4deg]" : "rotate-[0.4deg]"
   800→          } transition-transform hover:rotate-0 motion-reduce:rotate-0 motion-reduce:transition-none`}
   801→        >
   802→          <img
   803→            src={log.imagePath}
   804→            alt={`Foto latihan ${log.title}`}
   805→            loading="lazy"
   806→            className="w-full rounded-[2px] border border-foreground/15 bg-white p-1 shadow-[0_1px_4px_rgba(60,50,30,0.10)] transition-opacity hover:opacity-90"
   807→          />
   808→        </a>
   809→      )}
   810→      <div className="mt-3 flex justify-end gap-4">
   811→        <button type="button" onClick={startEdit} className={textLink}>
   812→          ubah
   813→        </button>
   814→        <button
   815→          type="button"
   816→          onClick={remove}
   817→          onBlur={() => setConfirming(false)}
   818→          className={`${textLink} ${confirming ? "text-destructive hover:text-destructive" : ""}`}
   819→        >
   820→          {confirming ? "yakin? klik lagi" : "hapus"}
   821→        </button>
   822→      </div>
   823→    </li>
   824→  );
   825→}

   829→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   830→async function downscaleImage(file: File): Promise<File | null> {
   831→  try {
   832→    const bitmap = await createImageBitmap(file);
   833→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   834→    const w = Math.round(bitmap.width * scale);
   835→    const h = Math.round(bitmap.height * scale);
   836→
   837→    const canvas = document.createElement("canvas");
   838→    canvas.width = w;
   839→    canvas.height = h;
   840→    const ctx = canvas.getContext("2d");
   841→    if (!ctx) return file;
   842→    ctx.drawImage(bitmap, 0, 0, w, h);
   843→    bitmap.close();
   844→
   845→    const blob = await new Promise<Blob | null>((resolve) =>
   846→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   847→    );
   848→    if (!blob) return file;
   849→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   850→      type: "image/jpeg",
   851→    });
   852→  } catch {
   853→    return file; // fall back to the original file
   854→  }
   855→}
