   262→function LogForm({ onCreated }: { onCreated: (log: PracticeLog) => void }) {

   265→  const [durationHours, setDurationHours] = useState("");

   271→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   272→  const [submitting, setSubmitting] = useState(false);
   273→  const [fieldError, setFieldError] = useState<string | null>(null);
   274→  const fileRef = useRef<HTMLInputElement>(null);

   285→    !isToday &&
   286→    todayKey !== "" &&
   287→    dateValue === toDateKey(new Date(Date.now() - 86400000));

   290→    dateValue && !isToday

   333→      fd.set("durationMin", String(minutes));

   336→
   337→      let file: File | null = photo;

   411→            Tanggal: {pickedDateLabel} ·{" "}

   547→function relativeDayLabel(day: string): string | null {
   548→  const today = toDateKey(new Date());
   549→  if (day === today) return "Hari ini";
   550→  const target = new Date(`${day}T00:00:00`).getTime();
   551→  const now = new Date(`${today}T00:00:00`).getTime();
   552→  const diffDays = Math.round((now - target) / 86400000);
   553→  if (diffDays === 1) return "Kemarin";
   554→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   555→  return null;
   556→}

   558→function LogRow({
   559→  log,
   560→  justAdded,
   561→  onDeleted,
   562→  onUpdated,
   563→}: {
   564→  log: PracticeLog;
   565→  justAdded: boolean;
   566→  onDeleted: (id: string) => void;
   567→  onUpdated: (log: PracticeLog) => void;
   568→}) {
   569→  const [confirming, setConfirming] = useState(false);
   570→  const [editing, setEditing] = useState(false);
   571→  const [saving, setSaving] = useState(false);
   572→  const [editError, setEditError] = useState<string | null>(null);
   573→  const [draft, setDraft] = useState({
   574→    actorName: log.actorName,
   575→    title: log.title,
   576→    durationHours: minutesToHoursInput(log.durationMin),
   577→    notes: log.notes,
   578→  });
   579→
   580→  const startEdit = () => {
   581→    setDraft({
   582→      actorName: log.actorName,
   583→      title: log.title,
   584→      durationHours: minutesToHoursInput(log.durationMin),
   585→      notes: log.notes,
   586→    });
   587→    setEditError(null);
   588→    setConfirming(false);
   589→    setEditing(true);
   590→  };
   591→
   592→  const saveEdit = async () => {
   593→    const name = draft.actorName.trim();
   594→    const title = draft.title.trim();
   595→    const minutes = hoursToMinutes(draft.durationHours);
   596→    if (!name) return setEditError("Nama tidak boleh kosong.");
   597→    if (!title) return setEditError("Judul tidak boleh kosong.");
   598→    if (minutes === null || minutes < 1) {
   599→      return setEditError("Isi durasi (jam).");
   600→    }
   601→    if (minutes > 1440) {
   602→      return setEditError("Durasi maksimal 24 jam.");
   603→    }
   604→
   605→    setSaving(true);
   606→    setEditError(null);
   607→    try {
   608→      const res = await fetch(`/api/logs/${log.id}`, {
   609→        method: "PATCH",
   610→        headers: { "Content-Type": "application/json" },
   611→        body: JSON.stringify({
   612→          actorName: name,
   613→          title,
   614→          durationMin: minutes,
   615→          notes: draft.notes.trim(),
   616→        }),
   617→      });
   618→      const data = (await res.json().catch(() => null)) as
   619→        | { log?: PracticeLog; error?: string }
   620→        | null;
   621→      if (!res.ok || !data?.log) {
   622→        throw new Error(data?.error ?? "Gagal menyimpan perubahan.");
   623→      }
   624→      onUpdated(data.log);
   625→      setEditing(false);
   626→      toast.success("Perubahan tersimpan.");
   627→    } catch (err) {
   628→      setEditError(err instanceof Error ? err.message : "Gagal menyimpan perubahan.");
   629→    } finally {
   630→      setSaving(false);
   631→    }
   632→  };
   633→
   634→  const remove = async () => {
   635→    if (!confirming) {
   636→      setConfirming(true);
   637→      return;
   638→    }
   639→    try {
   640→      const res = await fetch(`/api/logs/${log.id}`, { method: "DELETE" });
   641→      if (!res.ok) throw new Error();
   642→      onDeleted(log.id);
   643→      toast.success("Catatan dihapus.");
   644→    } catch {
   645→      toast.error("Gagal menghapus. Coba lagi.");
   646→      setConfirming(false);
   647→    }
   648→  };
   649→
   650→  const day = dateKey(log.date);
   651→  const dayLabel =
   652→    relativeDayLabel(day) ??
   653→    new Date(`${day}T00:00:00`).toLocaleDateString("id-ID", {
   654→      day: "numeric",
   655→      month: "short",
   656→    });
   657→
   658→  if (editing) {
   659→    return (
   660→      <li className="rounded-[3px] border border-foreground/30 bg-card p-4">
   661→        <div className="space-y-4">
   662→          <div className="grid gap-4 sm:grid-cols-2">
   663→            <input
   664→              value={draft.actorName}
   665→              onChange={(e) => setDraft({ ...draft, actorName: e.target.value })}
   666→              aria-label="Nama/Divisi"
   667→              className={underlineInput}
   668→              autoFocus
   669→            />
   670→            <input
   671→              value={draft.durationHours}
   672→              onChange={(e) => setDraft({ ...draft, durationHours: e.target.value })}
   673→              type="text"
   674→              inputMode="decimal"
   675→              autoComplete="off"
   676→              aria-label="Durasi (jam)"
   677→              className={`${underlineInput} tabular-nums`}
   678→            />
   679→          </div>
   680→          <input
   681→            value={draft.title}
   682→            onChange={(e) => setDraft({ ...draft, title: e.target.value })}
   683→            aria-label="Apa yang dilatih"
   684→            className={underlineInput}
   685→          />
   686→          <textarea
   687→            value={draft.notes}
   688→            onChange={(e) => setDraft({ ...draft, notes: e.target.value })}
   689→            rows={3}
   690→            aria-label="Catatan"
   691→            className="w-full resize-y rounded-[3px] border border-foreground/15 bg-transparent px-3 py-2 text-sm outline-none transition-colors focus:border-seal"
   692→          />
   693→          {editError && (
   694→            <p role="alert" className="text-sm text-destructive">
   695→              {editError}
   696→            </p>
   697→          )}
   698→          <div className="flex items-center justify-end gap-4">
   699→            <button type="button" onClick={() => setEditing(false)} className={textLink}>
   700→              batal
   701→            </button>
   702→            <button
   703→              type="button"
   704→              onClick={saveEdit}
   705→              disabled={saving}
   706→              className="rounded-sm text-xs font-semibold underline underline-offset-2 disabled:opacity-50 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ring"
   707→            >
   708→              {saving ? "menyimpan…" : "simpan"}
   709→            </button>
   710→          </div>
   711→        </div>
   712→      </li>
   713→    );
   714→  }
   715→
   716→  return (
   717→    <li
   718→      id={`log-${log.id}`}
   719→      className={`rounded-[3px] border bg-card p-4 transition-colors hover:border-foreground/30 ${
   720→        justAdded ? "entry-flash border-seal/40" : "border-foreground/10"
   721→      }`}
   722→    >
   723→      <div className="flex items-baseline justify-between gap-3">
   724→        <p className="font-serif text-sm font-bold">{log.actorName}</p>
   725→        <p className="shrink-0 text-xs tabular-nums text-muted-foreground">
   726→          {dayLabel} · {formatDuration(log.durationMin)}
   727→        </p>
   728→      </div>
   729→      <p className="mt-1 text-sm">{log.title}</p>
   730→      {log.notes && (
   731→        <p className="mt-1.5 whitespace-pre-line text-sm text-muted-foreground">{log.notes}</p>
   732→      )}
   733→      {log.imagePath && (
   734→        <a href={log.imagePath} target="_blank" rel="noreferrer" className="mt-3 block">
   735→          <img
   736→            src={log.imagePath}
   737→            alt={`Foto latihan ${log.title}`}
   738→            loading="lazy"
   739→            className="max-h-80 w-full rounded-[3px] border border-foreground/10 object-cover transition-opacity hover:opacity-90"
   740→          />
   741→        </a>
   742→      )}
   743→      <div className="mt-3 flex justify-end gap-4">
   744→        <button type="button" onClick={startEdit} className={textLink}>
   745→          ubah
   746→        </button>
   747→        <button
   748→          type="button"
   749→          onClick={remove}
   750→          onBlur={() => setConfirming(false)}
   751→          className={`${textLink} ${confirming ? "text-destructive hover:text-destructive" : ""}`}
   752→        >
   753→          {confirming ? "yakin? klik lagi" : "hapus"}
   754→        </button>
   755→      </div>
   756→    </li>
   757→  );
   758→}

   762→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   763→async function downscaleImage(file: File): Promise<File | null> {
   764→  try {
   765→    const bitmap = await createImageBitmap(file);
   766→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   767→    const w = Math.round(bitmap.width * scale);
   768→    const h = Math.round(bitmap.height * scale);
   769→
   770→    const canvas = document.createElement("canvas");
   771→    canvas.width = w;
   772→    canvas.height = h;
   773→    const ctx = canvas.getContext("2d");
   774→    if (!ctx) return file;
   775→    ctx.drawImage(bitmap, 0, 0, w, h);
   776→    bitmap.close();
   777→
   778→    const blob = await new Promise<Blob | null>((resolve) =>
   779→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   780→    );
   781→    if (!blob) return file;
   782→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   783→      type: "image/jpeg",
   784→    });
   785→  } catch {
   786→    return file; // fall back to the original file
   787→  }
   788→}
