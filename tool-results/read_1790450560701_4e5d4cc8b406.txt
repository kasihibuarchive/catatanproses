   296→   296→function LogForm({
   297→   297→  onCreated,
   298→   298→  lastLog,
   299→   299→}: {
   300→   300→  onCreated: (log: PracticeLog) => void;
   301→   301→  lastLog: PracticeLog | null;
   302→   302→}) {
   303→   303→  const [actorName, setActorName] = useState("");
   304→   304→  const [title, setTitle] = useState("");
   305→   305→  const [durationHours, setDurationHours] = useState("");
   306→   306→  const [notes, setNotes] = useState("");
   307→   307→  const [dateValue, setDateValue] = useState(""); // "" = today
   308→   308→  const [showDatePicker, setShowDatePicker] = useState(false);
   309→   309→  const [todayKey, setTodayKey] = useState("");
   310→   310→  const [photo, setPhoto] = useState<File | null>(null);
   311→   311→  const [photoPreview, setPhotoPreview] = useState<string | null>(null);
   312→   312→  const [submitting, setSubmitting] = useState(false);
   313→   313→  const [fieldError, setFieldError] = useState<string | null>(null);
   314→   314→  const fileRef = useRef<HTMLInputElement>(null);
   315→   315→
   316→   316→  // Prefill name from localStorage (client only).
   317→   317→  useEffect(() => {
   318→   318→    const saved = window.localStorage.getItem(ACTOR_NAME_STORAGE_KEY);
   319→   319→    if (saved) setActorName(saved);
   320→   320→    setTodayKey(toDateKey(new Date()));
   321→   321→  }, []);
   322→   322→
   323→   323→  const isToday = !dateValue || (todayKey !== "" && dateValue === todayKey);
   324→   324→  const isYesterday =
   325→   325→    !isToday &&
   326→   326→    todayKey !== "" &&
   327→   327→    dateValue === toDateKey(new Date(Date.now() - 86400000));
   328→   328→
   329→   329→  // "Use last practice" is only useful while the form is still blank.
   330→   330→  const canReuse = Boolean(lastLog) && !title && !durationHours && !notes;
   331→   331→  const reuseLast = () => {
   332→   332→    if (!lastLog) return;
   333→   333→    setTitle(lastLog.title);
   334→   334→    setDurationHours(minutesToHoursInput(lastLog.durationMin));
   335→   335→    setNotes(lastLog.notes);
   336→   336→  };
   337→   337→
   338→   338→  const pickedDateLabel =
   339→   339→    dateValue && !isToday
   340→   340→      ? new Date(`${dateValue}T00:00:00`).toLocaleDateString("id-ID", {
   341→   341→          day: "numeric",
   342→   342→          month: "long",
   343→   343→        })
   344→   344→      : "hari ini";
   345→   345→
   346→   346→  const pickPhoto = (file: File | null) => {
   347→   347→    if (photoPreview) URL.revokeObjectURL(photoPreview);
   348→   348→    setPhoto(file);
   349→   349→    setPhotoPreview(file ? URL.createObjectURL(file) : null);
   350→   350→  };
   351→   351→
   352→   352→  const reset = (keepName: boolean) => {
   353→   353→    if (!keepName) setActorName("");
   354→   354→    setTitle("");
   355→   355→    setDurationHours("");
   356→   356→    setNotes("");
   357→   357→    setDateValue("");
   358→   358→    setShowDatePicker(false);
   359→   359→    pickPhoto(null);
   360→   360→    if (fileRef.current) fileRef.current.value = "";
   361→   361→  };
   362→   362→
   363→   363→  const submit = async (e?: React.FormEvent) => {
   364→   364→    e?.preventDefault();
   365→   365→    setFieldError(null);
   366→   366→
   367→   367→    if (!actorName.trim()) return setFieldError("Isi nama/divisi dulu ya.");
   368→   368→    if (!title.trim()) return setFieldError("Tulis apa yang dilatih hari ini.");
   369→   369→    const minutes = hoursToMinutes(durationHours);
   370→   370→    if (minutes === null || minutes < 1) {
   371→   371→      return setFieldError("Isi durasi latihan (jam).");
   372→   372→    }
   373→   373→    if (minutes > 1440) {
   374→   374→      return setFieldError("Durasi maksimal 24 jam.");
   375→   375→    }
   376→   376→
   377→   377→    setSubmitting(true);
   378→   378→    try {
   379→   379→      const fd = new FormData();
   380→   380→      fd.set("actorName", actorName.trim());
   381→   381→      fd.set("title", title.trim());
   382→   382→      fd.set("durationMin", String(minutes));
   383→   383→      fd.set("notes", notes.trim());
   384→   384→      if (!isToday && dateValue) fd.set("date", dateValue);
   385→   385→
   386→   386→      let file: File | null = photo;
   387→   387→      if (file) {
   388→   388→        // Downscale in the browser so uploads stay small.
   389→   389→        file = await downscaleImage(file);
   390→   390→        if (file) fd.set("image", file);
   391→   391→      }
   392→   392→
   393→   393→      const res = await fetch("/api/logs", { method: "POST", body: fd });
   394→   394→      const data = (await res.json().catch(() => null)) as
   395→   395→        | { log?: PracticeLog; error?: string }
   396→   396→        | null;
   397→   397→
   398→   398→      if (!res.ok || !data?.log) {
   399→   399→        throw new Error(data?.error ?? "Gagal menyimpan. Coba lagi.");
   400→   400→      }
   401→   401→
   402→   402→      window.localStorage.setItem(ACTOR_NAME_STORAGE_KEY, actorName.trim());
   403→   403→      onCreated(data.log);
   404→   404→      reset(true);
   405→   405→      toast.success("Latihan tercatat.");
   406→   406→    } catch (err) {
   407→   407→      setFieldError(err instanceof Error ? err.message : "Gagal menyimpan. Coba lagi.");
   408→   408→    } finally {
   409→   409→      setSubmitting(false);
   410→   410→    }
   411→   411→  };
   412→   412→
   413→   413→  return (
   414→   414→    <form
   415→   415→      onSubmit={submit}
   416→   416→      onKeyDown={(e) => {
   417→   417→        if (e.key === "Enter" && (e.metaKey || e.ctrlKey)) {
   418→   418→          e.preventDefault();
   419→   419→          void submit();
   420→   420→        }
   421→   421→      }}
   422→   422→      className="pb-8 pt-10"
   423→   423→      noValidate
   424→   424→    >
   425→   425→      <p
   426→   426→        aria-hidden
   427→   427→        className="font-kanji text-xs tracking-[0.35em] text-seal"
   428→   428→      >
   429→   429→        {isToday ? "今日の稽古" : isYesterday ? "昨日の稽古" : "過去の稽古"}
   430→   430→      </p>
   431→   431→      <h1 className="mt-2 font-serif text-2xl tracking-tight">
   432→   432→        {isToday ? "Catat latihan hari ini" : `Catat latihan ${pickedDateLabel}`}
   433→   433→      </h1>
   434→   434→      <p className="mt-2 text-sm text-muted-foreground">
   435→   435→        Isi detilnya, tambahkan foto kalau ada, lalu kirim.
   436→   436→      </p>
   437→   437→      {canReuse && (
   438→   438→        <p className="mt-2 text-xs text-muted-foreground">
   439→   439→          Latihan terakhir: “{lastLog?.title}” ·{" "}
   440→   440→          <button type="button" onClick={reuseLast} className={textLink}>
   441→   441→            pakai lagi
   442→   442→          </button>
   443→   443→        </p>
   444→   444→      )}
   445→   445→
   446→   446→      {/* Date row — quiet by default, backdate on demand */}
   447→   447→      <div className="mt-4 min-h-9">
   448→   448→        {showDatePicker ? (
   449→   449→          <div className="flex flex-wrap items-center gap-3">
   450→   450→            <input
   451→   451→              type="date"
   452→   452→              value={dateValue}
   453→   453→              max={todayKey || undefined}

   605→   605→function relativeDayLabel(day: string): string | null {
   606→   606→  const today = toDateKey(new Date());
   607→   607→  if (day === today) return "Hari ini";
   608→   608→  const target = new Date(`${day}T00:00:00`).getTime();
   609→   609→  const now = new Date(`${today}T00:00:00`).getTime();
   610→   610→  const diffDays = Math.round((now - target) / 86400000);
   611→   611→  if (diffDays === 1) return "Kemarin";
   612→   612→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   613→   613→  return null;
   614→   614→}

   829→   829→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   830→   830→async function downscaleImage(file: File): Promise<File | null> {
   831→   831→  try {
   832→   832→    const bitmap = await createImageBitmap(file);
   833→   833→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   834→   834→    const w = Math.round(bitmap.width * scale);
   835→   835→    const h = Math.round(bitmap.height * scale);
   836→   836→
   837→   837→    const canvas = document.createElement("canvas");
   838→   838→    canvas.width = w;
   839→   839→    canvas.height = h;
   840→   840→    const ctx = canvas.getContext("2d");
   841→   841→    if (!ctx) return file;
   842→   842→    ctx.drawImage(bitmap, 0, 0, w, h);
   843→   843→    bitmap.close();
   844→   844→
   845→   845→    const blob = await new Promise<Blob | null>((resolve) =>
   846→   846→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   847→   847→    );
   848→   848→    if (!blob) return file;
   849→   849→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   850→   850→      type: "image/jpeg",
   851→   851→    });
   852→   852→  } catch {
   853→   853→    return file; // fall back to the original file
   854→   854→  }
   855→   855→}
