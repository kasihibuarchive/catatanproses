   307→   307→/** Quiet link that renders the month recap as a washi PNG poster and downloads it. */
   308→   308→function DownloadMonthImage({ month }: { month: MonthGroup }) {
   309→   309→  const [busy, setBusy] = useState(false);
   310→   310→
   311→   311→  const download = async () => {
   312→   312→    setBusy(true);
   313→   313→    try {
   314→   314→      const blob = await renderMonthSummaryBlob(month);
   315→   315→      const url = URL.createObjectURL(blob);
   316→   316→      const a = document.createElement("a");
   317→   317→      a.href = url;
   318→   318→      a.download = `kekiro-${month.key}.png`;
   319→   319→      document.body.appendChild(a);
   320→   320→      a.click();
   321→   321→      a.remove();
   322→   322→      window.setTimeout(() => URL.revokeObjectURL(url), 4000);
   323→   323→      toast.success("Gambar ringkasan diunduh.");
   324→   324→    } catch {
   325→   325→      toast.error("Gagal membuat gambar.");
   326→   326→    } finally {
   327→   327→      setBusy(false);
   328→   328→    }
   329→   329→  };
   330→   330→
   331→   331→  return (
   332→   332→    <button type="button" onClick={download} disabled={busy} className={textLink}>
   333→   333→      {busy ? "menyiapkan…" : "gambar"}
   334→   334→    </button>
   335→   335→  );

   404→   404→  const reuseLast = () => {

   708→   708→function relativeDayLabel(day: string): string | null {
   709→   709→  const today = toDateKey(new Date());
   710→   710→  if (day === today) return "Hari ini";
   711→   711→  const target = new Date(`${day}T00:00:00`).getTime();
   712→   712→  const now = new Date(`${today}T00:00:00`).getTime();
   713→   713→  const diffDays = Math.round((now - target) / 86400000);
   714→   714→  if (diffDays === 1) return "Kemarin";
   715→   715→  if (diffDays >= 2 && diffDays <= 6) return `${diffDays} hari lalu`;
   716→   716→  return null;

   735→   735→    actorName: log.actorName,

   959→   959→/** Downscale to max 1600px JPEG 0.85 in the browser. Returns null on failure. */
   960→   960→async function downscaleImage(file: File): Promise<File | null> {
   961→   961→  try {
   962→   962→    const bitmap = await createImageBitmap(file);
   963→   963→    const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
   964→   964→    const w = Math.round(bitmap.width * scale);
   965→   965→    const h = Math.round(bitmap.height * scale);
   966→   966→
   967→   967→    const canvas = document.createElement("canvas");
   968→   968→    canvas.width = w;
   969→   969→    canvas.height = h;
   970→   970→    const ctx = canvas.getContext("2d");
   971→   971→    if (!ctx) return file;
   972→   972→    ctx.drawImage(bitmap, 0, 0, w, h);
   973→   973→    bitmap.close();
   974→   974→
   975→   975→    const blob = await new Promise<Blob | null>((resolve) =>
   976→   976→      canvas.toBlob(resolve, "image/jpeg", 0.85)
   977→   977→    );
   978→   978→    if (!blob) return file;
   979→   979→    return new File([blob], file.name.replace(/\.\w+$/, "") + ".jpg", {
   980→   980→      type: "image/jpeg",
   981→   981→    });
   982→   982→  } catch {
   983→   983→    return file; // fall back to the original file
   984→   984→  }
   985→   985→}
